45def _atomic_write_text(target: Path, text: str) -> None: 

46 target.parent.mkdir(parents=True, exist_ok=True) 

47 fd, tmp = tempfile.mkstemp(dir=str(target.parent), prefix=".tmp-", text=True) 

48 try: 

49 with os.fdopen(fd, "w") as f: 

50 f.write(text) 

51 os.replace(tmp, target) 

52 except Exception: 

53 Path(tmp).unlink(missing_ok=True) 

54 raise

93def _sessions_lock_path() -> Path: 

94 # A dedicated lock file, not SESSIONS_FILE itself: _atomic_write_text 

95 # replaces sessions.json's inode on every save (tempfile + os.replace), so 

96 # an flock held on a file descriptor opened against the OLD inode would 

97 # stop protecting anything the moment a save happens underneath it. A 

98 # lock file that no writer ever replaces keeps the same inode for the 

99 # life of the process, which is what flock's exclusion actually needs. 

100 return paths.RELAY_DIR / ".sessions.lock"

103@contextlib.contextmanager 

104def _sessions_lock(): 

105 """Exclusive lock around a sessions.json read-modify-write cycle. 

106 

107 PR #74 review (fixing #72): _load_sessions/_save_sessions is an 

108 unlocked snapshot-of-the-whole-registry read-modify-write. Wiring 

109 touch_peer into every send/drain made that hot -- two real OS processes 

110 racing on it can silently revert each other's writes, including a 

111 concurrent register_peer/rebind_session (the exact failure mode of #71: 

112 a peer whose registration got reverted can no longer identify itself). 

113 

114 flock is advisory: it only excludes OTHER holders of the same lock, so 

115 every writer that can race the newly-hot touch_peer must take it too -- 

116 a lock held by touch_peer alone would not have closed the touch-vs- 

117 register race this exists to fix. Scoped to touch_peer, register_peer, 

118 and rebind_session (the three functions that mutate last_seen and are 

119 named in the review); set_parent/remove_peer/rename_peer share the same 

120 unlocked-RMW shape against the same file but are administrative, 

121 rarely-called ops -- rename_peer in particular interleaves its sessions 

122 mutation with heavy per-message-file IO across four directories, so 

123 locking it correctly deserves its own follow-up rather than being 

124 folded into this fix. 

125 """ 

126 lock_path = _sessions_lock_path() 

127 lock_path.parent.mkdir(parents=True, exist_ok=True) 

128 with open(lock_path, "w") as f: 

129 fcntl.flock(f, fcntl.LOCK_EX) 

130 try: 

131 yield 

132 finally: 

133 fcntl.flock(f, fcntl.LOCK_UN)

140def _check_no_cycle(name: str, parent: str, sessions: dict[str, dict]) -> None: 

141 """Walk upward from `parent` following .parent pointers; raise 

142 CycleDetected if the walk reaches `name` (would create a cycle) or if 

143 parent==name outright (self-parent, the degenerate 1-cycle). Bounded by 

144 len(sessions) iterations so a pre-existing corrupt cycle already on disk 

145 (e.g. from a hand-edited sessions.json) can't hang this walk -- that's 

146 defense against corrupt existing data, not a depth limit on new writes; 

147 see "Why no depth cap" in docs/superpowers/specs/ 

148 2026-07-15-general-peer-tree-design.md.""" 

149 if parent == name: 

150 raise CycleDetected(f"{name!r} cannot be its own parent") 

151 chain = [name, parent] 

152 current = parent 

153 for _ in range(len(sessions)): 

154 next_parent = sessions.get(current, {}).get("parent") 

155 if next_parent is None: 

156 return 

157 chain.append(next_parent) 

158 if next_parent == name: 

159 raise CycleDetected( 

160 f"setting {name!r}'s parent to {parent!r} would create a " 

161 f"cycle: {' → '.join(chain)}" 

162 ) 

163 current = next_parent

166def _resolve_parent(name: str, sessions: dict[str, dict], existing: dict | None, 

167 parent: str | None) -> str | None: 

168 """Resolve the `parent` a peer should be stored with. 

169 

170 Explicit --parent always wins (validated: must exist, must not create a 

171 cycle). Any registered peer is a valid target regardless of role -- 

172 role is no longer a structural gate, see docs/superpowers/specs/ 

173 2026-07-15-general-peer-tree-design.md. Otherwise reuse the peer's 

174 already-stored parent (rebind case -- no new edge introduced, skip the 

175 cycle check). Otherwise auto-default only if exactly one role=parent 

176 peer exists — ambiguity or absence is a hard error, never a silent 

177 guess. Auto-default stays scoped to role=parent peers specifically (the 

178 common single-coordinator convenience); it does NOT expand to "any 

179 peer with no other candidates," since that would silently guess a 

180 non-obvious interior node as often as not.""" 

181 if parent is not None: 

182 target = sessions.get(parent) 

183 if target is None: 

184 raise InvalidParent(f"--parent {parent!r} is not a registered peer") 

185 _check_no_cycle(name, parent, sessions) 

186 return parent 

187 if existing and existing.get("parent"): 

188 return existing["parent"] 

189 parent_names = sorted( 

190 n for n, d in sessions.items() if d.get("role") == "parent" and n != name 

191 ) 

192 if len(parent_names) == 1: 

193 _check_no_cycle(name, parent_names[0], sessions) 

194 return parent_names[0] 

195 if not parent_names: 

196 raise InvalidParent( 

197 f"no role=parent peer exists yet to pair {name!r} with — " 

198 "register the parent first, or pass --parent explicitly" 

199 ) 

200 raise AmbiguousParent( 

201 f"multiple parent peers exist ({', '.join(parent_names)}) — " 

202 f"pass --parent explicitly to disambiguate {name!r}" 

203 )

220def _reparent_conflict_message(name: str, existing: dict, new_parent: str, 

221 session_id: str | None = None) -> str: 

222 """Build the refusal text, choosing the remedy by WHO is asking (#89). 

223 

224 "To reattach the SAME peer, omit --parent" is correct from the session 

225 that owns the record and destructive from any other, because 

226 `register <name>` takes its subject from the CALLING session -- omitting 

227 --parent skips the only check that fired and repoints session_id onto 

228 whoever typed it. Printing it unconditionally handed every reader a way 

229 out that re-creates the collision just prevented, and the reader most 

230 likely to follow it is the one it is wrong for. 

231 

232 `peers set-parent` is offered to everyone: it names its target as an 

233 ARGUMENT rather than implicitly, so it means the same thing typed in any 

234 window. That property, not the semantics, is what makes a command safe to 

235 put in an error message. 

236 """ 

237 counts = _peer_message_counts(name) 

238 total = sum(counts.values()) 

239 existing_parent = existing.get("parent") 

240 parent_desc = (f"parent {existing_parent!r}" if existing_parent 

241 else "no parent (a root peer)") 

242 incumbent = existing.get("session_id") or "" 

243 caller_owns_record = bool(session_id) and session_id == incumbent 

244 

245 head = ( 

246 f"{name!r} is already registered with {parent_desc} " 

247 f"(registered_at={existing.get('registered_at')}, {total} message" 

248 f"{'s' if total != 1 else ''} held: inbox={counts['inbox']} " 

249 f"delivered={counts['delivered']} processed={counts['processed']}). " 

250 f"Registering {name!r} with --parent {new_parent!r} would silently " 

251 f"re-home it, orphaning that history from its current parent's view. " 

252 ) 

253 move = (f"To deliberately move it, run: downbeat peers set-parent " 

254 f"{name!r} {new_parent!r}") 

255 if caller_owns_record: 

256 return head + f"To reattach the SAME peer, omit --parent. {move}" 

257 return head + ( 

258 f"This session is not the one {name!r} is bound to (that is " 

259 f"{incumbent[:8] or 'unknown'}), so re-registering from here would " 

260 f"take the binding as well as the parent. {move}" 

261 )

264def _incumbent_liveness(existing: dict) -> bool | None: 

265 """Is the session currently holding this record demonstrably still alive? 

266 

267 True -- its recorded claude_pid is a live claude process AND the start 

268 time matches, so the pid has not been recycled. 

269 False -- demonstrably gone: process dead, no longer claude, or a different 

270 process wearing the same pid. 

271 None -- cannot be determined. Deliberately NOT folded into False: a record 

272 with no recorded pid proves nothing, and treating unknown as 

273 "gone" is how a permissive default gets mistaken for a check that 

274 ran and passed. 

275 """ 

276 pid = existing.get("claude_pid") 

277 if not pid: 

278 return None 

279 # Imported here rather than at module scope: the store is usable without 

280 # process introspection, and only this check needs it. 

281 from . import session as _session 

282 try: 

283 if not _session._process_is_claude(pid): 

284 return False 

285 stored_start = existing.get("claude_pid_start") 

286 if not stored_start: 

287 return None 

288 live_start = _session.process_start_time(pid) 

289 if live_start is None: 

290 return None 

291 return live_start == stored_start 

292 except Exception: 

293 return None

296def register_peer(name: str, session_id: str, cwd: str, role: str, 

297 claude_pid: int | None = None, 

298 claude_pid_start: str | None = None, 

299 parent: str | None = None) -> Peer: 

300 name = _validate_peer_name(name) 

301 with _sessions_lock(): 

302 sessions = _load_sessions() 

303 existing = sessions.get(name) 

304 registered_at = existing["registered_at"] if existing else now_iso() 

305 history = list(existing.get("session_id_history", [])) if existing else [] 

306 if existing and existing.get("session_id") and existing["session_id"] != session_id: 

307 if existing["session_id"] not in history: 

308 history.append(existing["session_id"]) 

309 resolved_parent = ( 

310 _resolve_parent(name, sessions, existing, parent) 

311 if role == "child" or parent is not None or (existing and existing.get("parent")) 

312 else None 

313 ) 

314 # An explicit --parent that disagrees with an existing peer's stored 

315 # parent is presumed reattachment territory turning destructive: the 

316 # caller may mean "this is the same peer, move it" (fine, but must be 

317 # said explicitly) or may mean "this name just happens to collide with 

318 # someone else's child" (in which case overwriting parent would silently 

319 # re-home the wrong peer -- issue #70). Refuse before any write. Equal to 

320 # the stored parent is not a conflict -- it's confirming the existing 

321 # pairing. No --parent at all (parent is None) is the plain reattach path 

322 # and must never be gated here. 

323 if existing is not None and parent is not None and existing.get("parent") != parent: 

324 raise PeerReparentConflict( 

325 _reparent_conflict_message(name, existing, parent, session_id) 

326 ) 

327 # The parent check above prevented a session takeover only BY ACCIDENT 

328 # (#89): it refuses on a parent mismatch and never asks whether this 

329 # write would repoint session_id onto a different, still-running 

330 # session. Incidental protection is not protection -- omitting --parent 

331 # routes around that check by design, and nothing recorded that it had 

332 # been doing this job. 

333 # 

334 # Checked here rather than in the CLI because the TUI's add-peer modal 

335 # calls register_peer too; guarding one caller is how half of a fix 

336 # ends up inert in production. 

337 if (existing is not None 

338 and existing.get("session_id") 

339 and existing["session_id"] != session_id): 

340 live = _incumbent_liveness(existing) 

341 if live is True: 

342 raise PeerSessionTakeover( 

343 f"{name!r} is bound to session " 

344 f"{existing['session_id'][:8]}, whose process is still " 

345 f"running (claude_pid={existing.get('claude_pid')}). " 

346 f"Registering {name!r} from this session would take that " 

347 f"identity while its owner is using it. If you mean to " 

348 f"move the binding on purpose, run: downbeat rebind " 

349 f"{name!r} --session-id {session_id}" 

350 ) 

351 if live is None: 

352 # Not a refusal, and not a pass either. Say so, so that the 

353 # absence of an error is not read as a check having succeeded. 

354 _log.warning( 

355 "register peer=%s: incumbent liveness unknown " 

356 "(claude_pid=%s), allowing rebind from %s to %s", 

357 name, existing.get("claude_pid"), 

358 existing["session_id"][:8], str(session_id)[:8], 

359 ) 

360 # Identity is assigned once. Re-registering an existing name is the same 

361 # peer reattaching, not a new one, so its id must carry over untouched. 

362 peer_id = (existing or {}).get("peer_id") or new_id() 

363 peer = Peer( 

364 name=name, session_id=session_id, cwd=cwd, role=role, 

365 registered_at=registered_at, last_seen=now_iso(), 

366 claude_pid=claude_pid, 

367 claude_pid_start=claude_pid_start, 

368 session_id_history=history, 

369 parent=resolved_parent, 

370 peer_id=peer_id, 

371 ) 

372 sessions[name] = peer.to_dict() 

373 _save_sessions(sessions) 

374 _log.info("register peer=%s session=%s role=%s parent=%s claude_pid=%s", 

375 name, session_id, role, resolved_parent, claude_pid) 

376 return peer

428def remove_peer(name: str) -> None: 

429 """Remove a peer and promote its children to its own parent. 

430 

431 Without the promotion the children's `parent` pointers dangle: they drop 

432 out of every acting-as / children_of view while still living on disk and 

433 accepting messages (#19). Reattaching each child to the removed node's 

434 parent -- grandparent promotion -- keeps the tree connected; removing a 

435 root (parent=None) leaves its children as roots. For any forest the store 

436 API can produce this cannot create a cycle: the new parent is an ancestor 

437 of the removed node and therefore of the orphans, and an ancestor is never 

438 a descendant. Runs for every caller (CLI gc, the TUI remove, GcStaleModal's 

439 sweep), so a multi-peer sweep stays a forest whatever order the peers come 

440 off in.""" 

441 sessions = _load_sessions() 

442 removed = sessions.pop(name, None) 

443 if removed is None: 

444 return 

445 grandparent = removed.get("parent") # None if the removed peer was a root 

446 # Two corrupt-input guards so removal HEALS a hand-edited sessions.json 

447 # instead of forwarding its damage (neither is reachable via the store 

448 # API, which gates every edge through _check_no_cycle): 

449 # - a grandparent that no longer exists (dangling pointer) -> root. 

450 # - promoting a child to ITSELF (a 2-cycle A<->B: removing A would 

451 # repoint B to grandparent B) -> root, per child. 

452 if grandparent is not None and grandparent not in sessions: 

453 grandparent = None 

454 for child_name, peer in sessions.items(): 

455 if peer.get("parent") == name: 

456 peer["parent"] = None if grandparent == child_name else grandparent 

457 _save_sessions(sessions)

498def rename_peer(old_name: str, new_name: str) -> Peer: 

499 """Rename a peer, migrating ALL of its on-disk identity. Resumable, not 

500 transactional: an interrupted run is completed by re-running with the same 

501 (old_name, new_name); individual files never tear. 

502 

503 - rewrites `from`/`to` on every message file across inbox/, delivered/, 

504 processed/, quarantine/ (both mail sent BY and TO the peer); 

505 - relocates each of the four per-peer message directories 

506 `<base>/<old_name>/` → `<base>/<new_name>/`; 

507 - re-keys the sessions.json entry (dict key + the entry's own `name` 

508 field) and repoints every OTHER peer whose `parent == old_name`; 

509 - updates groups.json membership. 

510 

511 Not transactional but **resumable**: a `.rename-in-progress.json` marker is 

512 written before the first move and cleared on success, so re-running the same 

513 (old_name, new_name) after an interrupted run is recognized as a resume and 

514 converges to the fully-renamed state (per-file writes go through the store's 

515 atomic write-then-replace, so a crash never tears an individual file). The 

516 marker also lets a fresh rename reject a `new_name` whose message directory 

517 already exists on disk (a stale removed-peer dir) without mistaking a 

518 legitimate resume for that collision. Chosen over true multi-file rollback 

519 deliberately (issue #40, Option B): the store's atomicity primitive is 

520 per-file, so real rollback would be new machinery. 

521 

522 Raises PeerNotFound (old_name absent and not a resume), PeerNameCollision 

523 (new_name already registered, or a stale on-disk dir), or InvalidPeerName. 

524 """ 

525 new_name = _validate_peer_name(new_name) 

526 sessions = _load_sessions() 

527 marker = _read_rename_marker() 

528 resuming = (marker is not None 

529 and marker.get("from") == old_name 

530 and marker.get("to") == new_name) 

531 

532 if old_name not in sessions: 

533 # old is gone. Only a genuine in-progress rename (matching marker) may 

534 # be finished here; otherwise old simply never existed. Without this 

535 # gate, `rename('typo', an_existing_peer)` silently "succeeds". 

536 if resuming and new_name in sessions: 

537 _rename_in_groups(old_name, new_name) 

538 _rename_marker().unlink(missing_ok=True) 

539 return Peer.from_dict(sessions[new_name]) 

540 raise PeerNotFound(old_name) 

541 if new_name == old_name: 

542 return Peer.from_dict(sessions[old_name]) 

543 if new_name in sessions: 

544 raise PeerNameCollision(new_name) 

545 

546 bases = (paths.INBOX_DIR, paths.DELIVERED_DIR, 

547 paths.PROCESSED_DIR, paths.QUARANTINE_DIR) 

548 

549 if not resuming: 

550 # A per-peer directory already existing for an unregistered new_name is 

551 # a stale removed-peer directory (remove_peer leaves message dirs on 

552 # disk); migrating into it would merge two peers' mail under one name. 

553 # A genuine resume (marker matches) legitimately has partial new dirs. 

554 for base in bases: 

555 if (base / new_name).exists(): 

556 raise PeerNameCollision( 

557 f"{new_name}: a message directory for this name already " 

558 "exists on disk (left by a removed peer); remove it first") 

559 

560 _atomic_write_text(_rename_marker(), 

561 json.dumps({"from": old_name, "to": new_name})) 

562 

563 # 1+2. Per message file: rewrite from/to (old→new), and relocate every file 

564 # that lives in old's own per-peer dir into new's dir. Idempotent — a second 

565 # pass finds nothing matching old, and a file already at the new path is 

566 # simply rewritten with identical content. 

567 for base in bases: 

568 if not base.exists(): 

569 continue 

570 for peer_dir in list(base.iterdir()): 

571 if not peer_dir.is_dir(): 

572 continue 

573 owned_by_old = peer_dir.name == old_name 

574 for f in list(peer_dir.glob("*.json")): 

575 try: 

576 msg = _read_message_at(f) 

577 except StoreCorrupt: 

578 # One unreadable file (even in an unrelated peer's dir, which 

579 # this full-tree scan must visit to catch from==old) must not 

580 # abort — and block every resume of — the whole rename. 

581 _log.warning("rename %s->%s: skipping unreadable %s", 

582 old_name, new_name, f) 

583 continue 

584 d = msg.to_dict() 

585 changed = False 

586 if d["from"] == old_name: 

587 d["from"] = new_name 

588 changed = True 

589 if d["to"] == old_name: 

590 d["to"] = new_name 

591 changed = True 

592 if owned_by_old: 

593 # Write to new dir first, then drop the old copy (same 

594 # write-then-unlink order as deliver_messages/ack_messages). 

595 dest = base / new_name / f.name 

596 _atomic_write_text(dest, Message.from_dict(d).to_json()) 

597 f.unlink() 

598 elif changed: 

599 _atomic_write_text(f, Message.from_dict(d).to_json()) 

600 old_dir = base / old_name 

601 if old_dir.exists() and old_dir.is_dir() and not any(old_dir.iterdir()): 

602 old_dir.rmdir() 

603 

604 # 3. sessions.json: re-key, fix the entry's own name, repoint parents. 

605 entry = sessions.pop(old_name) 

606 entry["name"] = new_name 

607 sessions[new_name] = entry 

608 for peer in sessions.values(): 

609 if peer.get("parent") == old_name: 

610 peer["parent"] = new_name 

611 _save_sessions(sessions) 

612 

613 # 4. groups.json membership. 

614 _rename_in_groups(old_name, new_name) 

615 

616 _rename_marker().unlink(missing_ok=True) 

617 _log.info("rename peer %s -> %s", old_name, new_name) 

618 return Peer.from_dict(sessions[new_name])

768def deliver_messages(peer_name: str, session_id: str, 

769 max: int = 20) -> list[Message]: 

770 """Move up to max messages from inbox/<peer>/ to delivered/<peer>/, 

771 stamping delivered_at + delivered_to_session_id.""" 

772 # Draining its own inbox is one of the two events that mean "this peer is 

773 # alive" (#72) -- touch unconditionally, including the empty/no-inbox 

774 # case: the peer asking for its mail is the participation signal, not 

775 # whether any happened to be waiting. Best-effort, same as send_message's 

776 # touch: a touch must never turn a successful drain into an error. 

777 try: 

778 touch_peer(peer_name) 

779 except PeerNotFound: 

780 pass 

781 inbox_dir = paths.INBOX_DIR / peer_name 

782 if not inbox_dir.exists(): 

783 return [] 

784 delivered_dir = paths.DELIVERED_DIR / peer_name 

785 delivered_dir.mkdir(parents=True, exist_ok=True) 

786 paths_sorted = sorted(inbox_dir.glob("*.json"), 

787 key=lambda p: p.stat().st_mtime)[:max] 

788 out: list[Message] = [] 

789 for p in paths_sorted: 

790 try: 

791 msg = _read_message_at(p) 

792 except StoreCorrupt: 

793 continue 

794 d = msg.to_dict() 

795 d["delivered_at"] = now_iso() 

796 d["delivered_to_session_id"] = session_id 

797 updated = Message.from_dict(d) 

798 target = delivered_dir / p.name 

799 _atomic_write_text(target, updated.to_json()) 

800 p.unlink() 

801 out.append(updated) 

802 _log.info("deliver msg=%s peer=%s to_session=%s redelivery=%d", 

803 msg.id, peer_name, session_id, msg.redelivery_count) 

804 _append_delivery_log({"event": "deliver", "msg_id": msg.id, 

805 "peer": peer_name, "session_id": session_id, 

806 "redelivery_count": msg.redelivery_count}) 

807 return out

810def ack_messages(ids: list[str]) -> dict[str, bool]: 

811 """For each id, find the message in delivered/, set delivery_ack_at, 

812 move to processed/, archive=True. Returns map of id→success.""" 

813 result: dict[str, bool] = {}

819 result[mid] = False 

820 continue

822 d = msg.to_dict() 

823 d["delivery_ack_at"] = now_iso() 

824 d["archived"] = True

826 peer = path.parent.name 

827 target = paths.PROCESSED_DIR / peer / path.name

832 result[mid] = True 

833 except Exception: 

834 _log.exception("ack failed for %s", mid) 

835 result[mid] = False 

836 return result

839def archive_messages(ids: list[str]) -> dict[str, bool]: 

840 """Move each message (in any of inbox/ delivered/) to processed/ with 

841 archived=True, auto-acking it if it was delivered-but-unacked. Used by the 

842 TUI "clear inbox" action to drain a peer's absorbed report-backlog without 

843 deleting anything (recoverable from processed/). Returns id→success.""" 

844 result: dict[str, bool] = {} 

845 for mid in ids: 

846 try: 

847 old = _find_message_path(mid) 

848 msg = _read_message_at(old) 

849 d = msg.to_dict() 

850 d["archived"] = True 

851 if d.get("delivered_at") is not None and d.get("delivery_ack_at") is None: 

852 d["delivery_ack_at"] = now_iso() 

853 updated = Message.from_dict(d) 

854 # Write the new copy BEFORE dropping the old one: the window this 

855 # leaves has the message in two places, which list_inbox's `seen` 

856 # set already dedupes. The reverse order left it in none -- see 

857 # the sibling movers (deliver_messages, ack_messages, reconcile, 

858 # requeue_quarantined), which all write-then-unlink. 

859 target = _message_path(updated) # archived → processed/ 

860 _write_message(updated) 

861 if old != target: # already-archived: same path, nothing to drop 

862 old.unlink() 

863 _log.info("archive msg=%s peer=%s", mid, updated.to_peer) 

864 _append_delivery_log({"event": "archive", "msg_id": mid, 

865 "peer": updated.to_peer}) 

866 result[mid] = True 

867 except Exception: 

868 _log.exception("archive failed for %s", mid) 

869 result[mid] = False 

870 return result

982def reconcile(window_minutes: int = 30, max_redelivery: int = 3) -> dict: 

983 """Scan delivered/. For each message with delivered_at older than 

984 window_minutes: auto-ack it if it is terminal (see _is_terminal_message), 

985 else requeue if redelivery_count < max_redelivery, else quarantine.""" 

986 from datetime import datetime, timedelta 

987 now = datetime.now(UTC) 

988 threshold = now - timedelta(minutes=window_minutes) 

989 counts = {"promoted": 0, "requeued": 0, "quarantined": 0, "auto_acked": 0} 

990 if not paths.DELIVERED_DIR.exists(): 

991 return counts 

992 for peer_dir in paths.DELIVERED_DIR.iterdir(): 

993 if not peer_dir.is_dir(): 

994 continue 

995 for p in list(peer_dir.glob("*.json")): 

996 try: 

997 msg = _read_message_at(p) 

998 except StoreCorrupt: 

999 continue 

1000 if not msg.delivered_at: 

1001 continue 

1002 try: 

1003 d_at = datetime.fromisoformat(msg.delivered_at) 

1004 except ValueError: 

1005 continue 

1006 if d_at > threshold: 

1007 continue # still within window 

1008 if _is_terminal_message(msg): 

1009 # Nothing is owed back — retire it instead of redelivering. 

1010 reason = ("reply" if msg.in_reply_to is not None 

1011 else f"terminal kind {msg.kind!r}") 

1012 if ack_messages([msg.id]).get(msg.id): 

1013 counts["auto_acked"] += 1 

1014 _log.info("auto-ack msg=%s peer=%s reason=%s", 

1015 msg.id, peer_dir.name, reason) 

1016 _append_delivery_log({"event": "auto-ack", 

1017 "msg_id": msg.id, 

1018 "peer": peer_dir.name, 

1019 "reason": reason}) 

1020 else: 

1021 _log.warning("auto-ack failed msg=%s peer=%s", 

1022 msg.id, peer_dir.name) 

1023 continue 

1024 if msg.redelivery_count + 1 > max_redelivery: 

1025 # Quarantine 

1026 d = msg.to_dict() 

1027 d["quarantined_at"] = now_iso() 

1028 d["quarantine_reason"] = ( 

1029 f"unacked after {max_redelivery} redeliveries" 

1030 ) 

1031 quarantined = Message.from_dict(d) 

1032 target = paths.QUARANTINE_DIR / peer_dir.name / p.name 

1033 target.parent.mkdir(parents=True, exist_ok=True) 

1034 _atomic_write_text(target, quarantined.to_json()) 

1035 p.unlink() 

1036 counts["quarantined"] += 1 

1037 _log.warning("quarantine msg=%s peer=%s reason=%s", 

1038 msg.id, peer_dir.name, d["quarantine_reason"]) 

1039 _append_delivery_log({"event": "quarantine", 

1040 "msg_id": msg.id, "peer": peer_dir.name}) 

1041 else: 

1042 # Requeue back to inbox/ 

1043 d = msg.to_dict() 

1044 d["delivered_at"] = None 

1045 d["delivered_to_session_id"] = None 

1046 d["redelivery_count"] = msg.redelivery_count + 1 

1047 requeued = Message.from_dict(d) 

1048 target = paths.INBOX_DIR / peer_dir.name / p.name 

1049 target.parent.mkdir(parents=True, exist_ok=True) 

1050 _atomic_write_text(target, requeued.to_json()) 

1051 p.unlink() 

1052 counts["requeued"] += 1 

1053 _log.info("requeue msg=%s peer=%s attempt=%d", 

1054 msg.id, peer_dir.name, d["redelivery_count"]) 

1055 _append_delivery_log({"event": "requeue", 

1056 "msg_id": msg.id, "peer": peer_dir.name, 

1057 "redelivery_count": d["redelivery_count"]}) 

1058 return counts

1077def migrate_store(dry_run: bool = False) -> dict: 

1078 """Eagerly upgrade every message file to CURRENT_SCHEMA_VERSION. 

1079 

1080 Upgrade-on-read already keeps anything the store touches current; this is 

1081 the flush for files nothing reads again (old processed/ archives, 

1082 quarantined mail), which would otherwise stay on an old version forever. 

1083 Rewrites each file in place — never at _message_path(), which would move a 

1084 message between directories as a side effect of a migration. 

1085 """ 

1086 # `scanned` is counted explicitly rather than derived by summing the 

1087 # others: a file can need both a version bump and an id backfill, or 

1088 # neither category (id-only), so any sum over a subset silently lies. 

1089 counts = {"scanned": 0, "migrated": 0, "current": 0, "unreadable": 0, 

1090 "ids_backfilled": 0} 

1091 # Resolving names to ids needs the peer registry, which a migration rung 

1092 # cannot reach (rungs are pure dict->dict). That is why this half of the 

1093 # v1->v2 migration lives here rather than in the ladder. 

1094 ids_by_name = {p.name: p.peer_id for p in list_peers() if p.peer_id} 

1095 for base in _message_dirs(): 

1096 if not base.exists(): 

1097 continue 

1098 for p in sorted(base.glob("*/*.json")): 

1099 counts["scanned"] += 1 

1100 try: 

1101 raw = json.loads(p.read_text()) 

1102 except (OSError, json.JSONDecodeError): 

1103 counts["unreadable"] += 1 

1104 continue 

1105 if not isinstance(raw, dict): 

1106 counts["unreadable"] += 1 

1107 continue 

1108 needs_version = raw.get("schema_version") != CURRENT_SCHEMA_VERSION

1121 continue

1123 # A name with no live peer stays unresolved rather than being 

1124 # given an invented identity; the name fallback carries it.

1126 from_peer_id=msg.from_peer_id or new_from, 

1127 to_peer_id=msg.to_peer_id or new_to) 

1128 counts["ids_backfilled"] += 1 

1129 if needs_version: 

1130 counts["migrated"] += 1

1175def broadcast_status(broadcast_id: str) -> list[dict]: 

1176 """Return one row per original target. State derived from sibling messages 

1177 and any reply messages that carry the same broadcast_id.""" 

1178 all_msgs = _scan_all_messages() 

1179 siblings = [m for m in all_msgs if m.broadcast_id == broadcast_id] 

1180 if not siblings: 

1181 return [] 

1182 # The original fan-out: each target appears as a recipient of one sibling 

1183 # with state != REPLY. A reply is a message FROM the target carrying the 

1184 # same broadcast_id. 

1185 originals = {m.to_peer: m for m in siblings 

1186 if m.from_peer != m.to_peer and not _is_reply(m, siblings)} 

1187 rows: list[dict] = [] 

1188 for target, original in originals.items(): 

1189 replies = [m for m in siblings 

1190 if m.from_peer == target and m.id != original.id] 

1191 if replies: 

1192 state = "replied" 

1193 elif original.state == MessageState.READ: 

1194 state = "read" 

1195 else: 

1196 state = "pending" 

1197 rows.append({"target": target, "state": state, 

1198 "original_id": original.id, 

1199 "reply_ids": [r.id for r in replies]}) 

1200 return rows

1203def rebind_session(name: str, new_session_id: str | None = None) -> Peer: 

1204 """Update only the session_id (and last_seen) for an existing peer. 

1205 role, cwd, registered_at are preserved. If new_session_id is None, the 

1206 function auto-detects via session.detect_session_id(); raises RelayError 

1207 if no detection is possible. 

1208 Also appends to rebind_log.jsonl and updates session_id_history.""" 

1209 from . import session as session_mod 

1210 from .errors import RelayError 

1211 

1212 with _sessions_lock(): 

1213 sessions = _load_sessions() 

1214 if name not in sessions: 

1215 raise PeerNotFound(name) 

1216 

1217 if new_session_id is None: 

1218 new_session_id = session_mod.detect_session_id() 

1219 if new_session_id is None: 

1220 raise RelayError( 

1221 "could not auto-detect a session id; pass --session-id " 

1222 "explicitly" 

1223 ) 

1224 

1225 entry = sessions[name] 

1226 old_sid = entry.get("session_id") 

1227 history = list(entry.get("session_id_history", [])) 

1228 if old_sid and old_sid != new_session_id and old_sid not in history: 

1229 history.append(old_sid) 

1230 entry["session_id"] = new_session_id 

1231 entry["session_id_history"] = history 

1232 entry["last_rebind_at"] = now_iso() 

1233 entry["last_seen"] = now_iso() 

1234 sessions[name] = entry 

1235 _save_sessions(sessions) 

1236 _log.info("rebind peer=%s old_session=%s new_session=%s", 

1237 name, old_sid, new_session_id) 

1238 # Append to rebind_log.jsonl 

1239 _append_rebind_log({"peer": name, "old_session_id": old_sid, 

1240 "new_session_id": new_session_id}) 

1241 return Peer.from_dict(entry)
